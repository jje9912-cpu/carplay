Supabase SQL Editor 실행용 테이블 생성 SQL (작업내역 컬럼 포함)

Supabase 연동 시 SQL Editor에 아래 코드를 붙여넣고 Run을 한 번만 눌러주세요.

create table if not exists service_ledgers (
  id bigint primary key,
  company text not null,             -- 소속회사 (예: 정직한놈들)
  customer text not null,            -- 고객명 (예: 김병희)
  phone text default '',             -- 전화번호 (예: 010-2345-6789)
  car_num text not null,             -- 차량번호 (예: 12머1734)
  service_date text not null,        -- 정비일자 (YYYY-MM-DD)
  work_content text,                 -- 작업내역
  labor numeric default 0,           -- 공임금액
  deposit numeric default 0,         -- 입금액
  method text default '미수',         -- 입금방법 (미수, 계좌이체, 카드, 현금)
  car_model text default '',         -- 차종 (예: 싼타페)
  created_at timestamp with time zone default timezone('utc'::text, now())
);
alter table service_ledgers enable row level security;
create policy "Public Access" on service_ledgers for all using (true) with check (true);

🛠️ 작업내역 상세 수정 (입력창 크기 조절 가능)

💡 입력창 우측 하단 모서리를 마우스로 잡고 당기면 입력 칸 크기를 원하는 대로 넓히거나 늘릴 수 있습니다.
